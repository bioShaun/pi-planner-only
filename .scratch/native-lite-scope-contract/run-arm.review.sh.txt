# BATCH_ROOT=/project/tmp/pi-native-lite-<unique timestamp>; all inputs and
# outputs must already be frozen. GATE_LOG_ID must be unique for this invocation.
: "${BATCH_ID:?required}" "${BATCH_ROOT:?required}" "${ATTEMPT:?required}"
: "${ARM:?required}" "${GATE_LOG_ID:?required}" "${ARM_DEADLINE_MS:?required}" "${ARM_COST_STOP_USD:?required}"
case "$ARM" in S1-native|S1-lite|S2-lite|S2-native) ;; *) exit 1 ;; esac
case "$ARM_DEADLINE_MS" in *[!0-9]*|'') exit 1 ;; esac
export TMPDIR=/project/tmp/pi-native-lite-scope-next/tmp
export TMP="$TMPDIR" TEMP="$TMPDIR"
printf 'arm=%s deadlineMs=%s reportedCostStopUsd=%s\n' "$ARM" "$ARM_DEADLINE_MS" "$ARM_COST_STOP_USD" > "$BATCH_ROOT/logs/$GATE_LOG_ID.limits"
cd "$BATCH_ROOT/worktrees/$ARM" || exit 1
/home/tcuni-claw/.local/bin/slot audit > "$BATCH_ROOT/logs/$GATE_LOG_ID.audit" 2>&1 || exit 1
/home/tcuni-claw/.local/bin/slot status > "$BATCH_ROOT/logs/$GATE_LOG_ID.status" 2>&1 || exit 1
# Root must inspect audit/status and explicitly confirm the same launch gate
# remains valid after these checks, before invoking the command below.

[ "$confirmation" = LAUNCH ] || exit 1

  PI_NATIVE_LITE_BATCH_ROOT="$BATCH_ROOT" PI_NATIVE_LITE_BATCH_ID="$BATCH_ID" \

  PI_NATIVE_LITE_RPC_DEADLINE_MS="$ARM_DEADLINE_MS" TMPDIR="$TMPDIR" TMP="$TMP" TEMP="$TEMP" \
  node --experimental-strip-types /home/tcuni-claw/pi/pi-planner-only/.scratch/native-lite-lifecycle/run-rpc-arm.mjs
